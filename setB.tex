\noindent\textbf{1.} $\sqrt{(-7)^{2}}$의 제곱근은? \scorebox{3}

\begin{enumerate}[label=\protect\choicenum{\arabic*},leftmargin=*,itemsep=0pt]
  \item $\pm\,7$
  \item $\pm\,\sqrt{7}$
  \item $7$
  \item $-7$
  \item $49$
\end{enumerate}

\vspace{3mm}

\noindent\textbf{2.} 실수의 대소관계로 옳은 것은? \scorebox{3}

\begin{enumerate}[label=\protect\choicenum{\arabic*},leftmargin=*,itemsep=0pt]
  \item $\sqrt{10} > 4$
  \item $\sqrt{3} < 1$
  \item $-\sqrt{10} > -3$
  \item $-\sqrt{26} < -5$
  \item $\dfrac{\sqrt{2}}{3} > \dfrac{\sqrt{3}}{3}$
\end{enumerate}
